### 2.1 数据访问层 `internal/model/customer.go`

```go
package model

import (
	"go-fin-server/internal/db"
	"go-fin-server/pkg/types"
	"go-fin-server/pkg/utils"
)

// Customer 客户实体
type Customer struct {
	Id      uint64          `gorm:"column:id;primary_key" json:"id" description:"客户ID"`
	Name    string          `gorm:"column:name" json:"name" description:"客户名称" binding:"required" err_msg:"客户名称 必填"`
	Phone   string          `gorm:"column:phone" json:"phone" description:"联系电话"`
	Level   string          `gorm:"column:level;default:'0'" json:"level" description:"客户等级;0:普通,1:重要,2:VIP"`
	Remark  string          `gorm:"column:remark" json:"remark" description:"备注"`
	Status  string          `gorm:"column:status;default:'0'" json:"status" description:"状态;0:禁用,1:正常"`
	BaseModel
}

func (c Customer) TableName() string { return "customer" }

// SelectCustomerList 分页查询客户列表
func (c Customer) SelectCustomerList(conditions map[string]interface{}, sortConditions []string, pageNum, pageSize int) ([]Customer, int64, error) {
	query := db.DBConnections["master"].Model(&Customer{})
	offset, limit := utils.PageParse(pageNum, pageSize)
	for key, value := range conditions {
		query = query.Where(key, value)
	}
	data := make([]Customer, 0)
	var count int64
	if err := query.Count(&count).Error; err != nil {
		return data, count, err
	}
	for _, condition := range sortConditions {
		query = query.Order(condition)
	}
	d := query.Offset(offset).Limit(limit).Find(&data)
	return data, count, d.Error
}

// SelectCustomerById 根据ID查询客户
func (c Customer) SelectCustomerById(id uint64) (Customer, error) {
	var data Customer
	d := db.DBConnections["master"].Model(&Customer{}).Where("id = ?", id).First(&data)
	return data, d.Error
}

// InsertCustomer 新增客户
func (c Customer) InsertCustomer(data *Customer) error {
	return db.DBConnections["master"].Create(data).Error
}

// UpdateCustomer 修改客户
func (c Customer) UpdateCustomer(data *Customer) error {
	return db.DBConnections["master"].Model(&Customer{}).Where("id = ?", data.Id).Updates(data).Error
}

// DeleteCustomerByIds 批量删除客户
func (c Customer) DeleteCustomerByIds(ids []uint64) error {
	return db.DBConnections["master"].Where("id IN ?", ids).Delete(&Customer{}).Error
}
```

### 2.4 处理器层 `internal/handler/system/customer.go`

```go
package system

import (
	"go-fin-server/api"
	"go-fin-server/internal/model"
	"go-fin-server/internal/response"
	"go-fin-server/internal/service"
	"go-fin-server/pkg"
	"go-fin-server/pkg/utils"
	"strconv"
	"strings"

	"github.com/gin-gonic/gin"
)

type CustomerHandler struct {
	Services service.Services
	Api      api.Api
}

func NewCustomerHandler() *CustomerHandler {
	return &CustomerHandler{}
}

// ListCustomer 查询客户列表
//
//	@Summary	查询客户列表
//	@Tags		客户管理
//	@Produce	json
//	@Param		name		query	string	false	"客户名称"
//	@Param		pageNum		query	int		false	"页码"
//	@Param		pageSize	query	int		false	"每页数量"
//	@Success	200			{object}	map[string]interface{}
//	@Router		/system/customer/list [get]
//	@Security	BearerAuth
func (s *CustomerHandler) ListCustomer(c *gin.Context) {
	response.SetOperTitle(c, "查询客户列表")
	req := s.Api.Customer.CustomerListRequest
	if err := api.ShouldBindError(c, &req); err != nil {
		response.Error(c, err.Error())
		return
	}
	conditions := utils.BuildConditions(req)
	sortConditions := utils.BuildSortConditions(req.PageNum) // 排序字段按需调整
	data, total, err := s.Services.CustomerService.SelectCustomerList(conditions, sortConditions, req.PageNum, req.PageSize)
	if err != nil {
		pkg.Logger.Error(err)
		response.Error(c, err.Error())
		return
	}
	response.PageData(c, data, total)
}

// GetCustomer 查询客户详情
//
//	@Summary	查询客户详情
//	@Tags		客户管理
//	@Param		id	path	int	true	"客户ID"
//	@Success	200	{object}	map[string]interface{}
//	@Router		/system/customer/{id} [get]
//	@Security	BearerAuth
func (s *CustomerHandler) GetCustomer(c *gin.Context) {
	response.SetOperTitle(c, "查询客户详情")
	req := s.Api.Customer.CustomerIdUriRequest
	if err := api.ShouldBindUriError(c, &req); err != nil {
		response.Error(c, err.Error())
		return
	}
	data, err := s.Services.CustomerService.SelectCustomerById(*req.Id)
	if err != nil {
		pkg.Logger.Error(err)
		response.Error(c, err.Error())
		return
	}
	response.Data(c, data)
}

// AddCustomer 新增客户
func (s *CustomerHandler) AddCustomer(c *gin.Context) {
	response.SetOperTitle(c, "新增客户")
	var req model.Customer
	if err := api.ShouldBindError(c, &req); err != nil {
		response.Error(c, err.Error())
		return
	}
	if err := s.Services.CustomerService.InsertCustomer(&req); err != nil {
		pkg.Logger.Error(err)
		response.Error(c, err.Error())
		return
	}
	response.DataMsg(c, nil, "新增成功")
}

// UpdateCustomer 修改客户
func (s *CustomerHandler) UpdateCustomer(c *gin.Context) {
	response.SetOperTitle(c, "修改客户")
	var req model.Customer
	if err := api.ShouldBindError(c, &req); err != nil {
		response.Error(c, err.Error())
		return
	}
	if err := s.Services.CustomerService.UpdateCustomer(&req); err != nil {
		pkg.Logger.Error(err)
		response.Error(c, err.Error())
		return
	}
	response.DataMsg(c, nil, "修改成功")
}

// DelCustomer 批量删除客户
func (s *CustomerHandler) DelCustomer(c *gin.Context) {
	response.SetOperTitle(c, "删除客户")
	idsStr := strings.Split(c.Param("ids"), ",")
	ids := make([]uint64, 0, len(idsStr))
	for _, idStr := range idsStr {
		if id, err := strconv.ParseUint(idStr, 10, 64); err == nil {
			ids = append(ids, id)
		}
	}
	if err := s.Services.CustomerService.DeleteCustomerByIds(ids); err != nil {
		pkg.Logger.Error(err)
		response.Error(c, err.Error())
		return
	}
	response.DataMsg(c, nil, "删除成功")
}
```

### 2.5 路由层 `internal/router/system/customer.go`

```go
package system

import (
	"go-fin-server/internal/handler/system"
	"go-fin-server/internal/middleware"

	"github.com/gin-gonic/gin"
)

func SetupCustomerRoutes(g *gin.RouterGroup) {
	customerHandler := system.NewCustomerHandler()
	customerGroup := g.Group("/customer")
	{
		customerGroup.GET("list", middleware.PermissionMiddleware("system:customer:list"), customerHandler.ListCustomer)   // 查询客户列表
		customerGroup.GET(":id", middleware.PermissionMiddleware("system:customer:query"), customerHandler.GetCustomer)  // 查询客户详情
		customerGroup.POST("", middleware.PermissionMiddleware("system:customer:add"), customerHandler.AddCustomer)      // 新增客户
		customerGroup.PUT("", middleware.PermissionMiddleware("system:customer:edit"), customerHandler.UpdateCustomer)   // 修改客户
		customerGroup.DELETE("/:ids", middleware.PermissionMiddleware("system:customer:remove"), customerHandler.DelCustomer) // 批量删除
	}
}
```

**注册路由**：在 `internal/router/router.go` 中，于 `system` 路由组内调用 `SetupCustomerRoutes(...)`。

## 五、布尔字段（checkbox 开关）约定示例

以角色管理的"菜单/部门树是否关联显示"为例（`menuCheckStrictly`、`deptCheckStrictly`），演示**是/否类字段**从数据库到前后端的完整布尔约定：前端 checkbox/开关直接绑定布尔，后端用 `bool` 接收，数据库用 `BOOLEAN` 存储，全程**无需转 `0/1`**。

### 5.1 数据库列定义（`server/data/ry_init_full.sql`）

```sql
CREATE TABLE `sys_role` (
  ...
  `menu_check_strictly` BOOLEAN DEFAULT 1 COMMENT '菜单树选择项是否关联显示（布尔，对接前端 checkbox）',
  `dept_check_strictly` BOOLEAN DEFAULT 1 COMMENT '部门树选择项是否关联显示（布尔，对接前端 checkbox）',
  ...
) ;
```

> `BOOLEAN` 即 MySQL 的 `tinyint(1)`，存 `0/1`，与 Go `bool`、前端 `boolean` 一一对应。

### 5.2 后端模型字段（`server/internal/model/sys_role.go`）

```go
type SysRole struct {
	Id               uint64 `gorm:"column:id;primary_key" json:"id"`
	RoleName         string `gorm:"column:role_name" json:"roleName"`
	MenuCheckStrictly bool   `gorm:"column:menu_check_strictly;default:1" json:"menuCheckStrictly"`
	DeptCheckStrictly bool   `gorm:"column:dept_check_strictly;default:1" json:"deptCheckStrictly"`
	// ...
}
```

### 5.3 后端请求结构（`server/api/system/sys_role.go`）

```go
type UpdateRoleRequest struct {
	Id                uint64 `gorm:"column:id;primary_key;" json:"id"`
	RoleName          string `gorm:"column:role_name" json:"roleName"`
	MenuCheckStrictly bool   `gorm:"column:menu_check_strictly" json:"menuCheckStrictly"`
	DeptCheckStrictly bool   `gorm:"column:dept_check_strictly" json:"deptCheckStrictly"`
	// ...
}
```

> 后端字段必须是 `bool`：gin 的 JSON 绑定才能接受前端 `true/false`。若误用 `uint`，提交会报 `Invalid type for field menuCheckStrictly, expected uint`。

### 5.4 后端树查询函数参数（`sys_menu.go` / `sys_role.go`）

### 5.7 前端提交（直接传布尔，不转 0/1）

```ts
// api.add / api.update
add: (data) => {
  data.menuIds = getMenuAllCheckedKeys()
  return addRole(data) // data.menuCheckStrictly 保持布尔 true/false
},
update: (data) => {
  data.menuIds = getMenuAllCheckedKeys()
  return updateRole(data)
}
```

> 后端为 `bool` 时**务必不要**把布尔转成 `0/1` 数字提交——gin 绑定时 `1` 无法解析为 `bool`，同样会报绑定错误。布尔字段提交时原样传 `true/false` 即可。

### 5.8 注意事项小结

- 是/否字段：前端 `boolean` ↔ 后端 `bool` ↔ 数据库 `BOOLEAN(tinyint(1))`，三层一致，不转数字。
- 若某个 checkbox 字段后端仍为 `uint`（历史遗留），才需要前端转 `0/1`；统一布尔化后无需转换。
